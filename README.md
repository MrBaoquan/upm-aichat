    // 示例代码
    var _content = string.Empty;
    AIChatManager
        .ChatStream(
            $"签文内容为:天开地辟作良缘，日吉时良万物全；若得此签非小可，公行忠正帝王宣。。为我提供财运方面的解签, 我是女性, 生日为2000-02-01",
            @"  你是一名解签助手，你的任务是根据用户的输入，为用户提供解签内容。
                你需要计算用户年龄，对于未成年，要避免出现财运、姻缘等不适合的内容，拒绝为用户提供解签。
                要求解签内容必须是纯文本，且内容数量严格限制在40-50字！不要出现任何关于字数相关的统计信息！
            "
        )
        .ObserveOnMainThread()
        .Subscribe(
            _msg =>
            {
                _content += _msg;
            },
            () =>
            {
                Debug.Log(_content);
            }
        );
